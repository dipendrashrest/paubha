## Usage

```tsx
import {
  ActivityFeed,
  ActivityFeedItem,
} from "@/components/ui/activity-feed/activity-feed";
import { Avatar } from "@/components/ui/avatar/avatar";

<ActivityFeed>
  <ActivityFeedItem
    avatar={<Avatar initials="AC" alt="Anna Chen" size="sm" />}
    title={
      <>
        <span className="font-semibold">Anna Chen</span> commented on Project
        Atlas
      </>
    }
    timestamp="2h ago"
  />
</ActivityFeed>
```
