## Usage

```tsx
import { ActiveFilters, FilterBar, FilterChip } from "@/components/ui/filter";

<FilterBar>
  <FilterChip label="Status" selected />
  <FilterChip label="Priority" />
</FilterBar>
<ActiveFilters onClear={() => {}}>
  <FilterChip label="Assignee: Sarah" selected onRemove={() => {}} />
</ActiveFilters>
```
