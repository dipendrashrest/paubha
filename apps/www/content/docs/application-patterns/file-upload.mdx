## Usage

```tsx
import { FileUpload, FileUploadItem, FileUploadList } from "@/components/ui/file-upload";

<FileUpload onFilesChange={(files) => console.log(files)} />
<FileUploadList>
  <FileUploadItem name="brief.pdf" size="1.2 MB" progress={40} />
</FileUploadList>
```
