## Usage

```tsx
import { BarChart, ChartCard } from "@/components/ui/chart";

<ChartCard title="Revenue">
  <BarChart
    data={[
      { label: "Mon", value: 40 },
      { label: "Tue", value: 65 },
      { label: "Wed", value: 52 },
    ]}
  />
</ChartCard>
```
